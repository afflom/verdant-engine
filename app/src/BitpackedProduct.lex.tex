definition{
  name: bitDivisor,
  parameters: [shift : Nat],
  result: Nat,
  body: (if (shift =? 0) then 1 else (if (shift =? 1) then 2 else (if (shift =? 2) then 4 else (if (shift =? 3) then 8 else (if (shift =? 4) then 16 else (if (shift =? 5) then 32 else (if (shift =? 6) then 64 else 128))))))),

definition{
  name: evaluate,
  parameters: [request : Request],
  result: Step,
  body: (match request
    | Request.layout two bytes blocks => (let groups : Nat := (if two then 2 else 4); (if (((prim remainder bytes 18 0 : Nat) =? 0) && ((prim multiply (prim quotient bytes 18 0 : Nat) groups : Nat) =? blocks)) then (ctor Step.row 18 groups (if two then 8 else 4) (if two then 4 else 8)) else (ctor Step.refuse)))
    | Request.coefficient two byte lane => (if ((byte <? 256) && (lane <? (if two then 4 else 8))) then (match (PackedWord::extract byte 1 (bitDivisor (if two then (prim multiply lane 2 : Nat) else lane)) (if two then 4 else 2))
      | Option.some raw => (ctor Step.value (if two then raw else (prim multiply raw 2 : Nat)))
      | Option.none => (ctor Step.refuse)) else (ctor Step.refuse))),
  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
}

  name: lastSignBit,
