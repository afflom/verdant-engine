  name: blockScale,

  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
}

  name: groupScale,
  parameters: [value : Nat, inverse : Nat, base : Nat, limit : Nat],
  result: GroupScale,
  body: (let scaled : Nat := (ScalarF32::multiply value inverse); (let code : Nat := (roundedLevel scaled limit 1262485504); (record GroupScale {code := code, decoded := (ScalarF32::multiply base (PackedWord::magnitudeBits code))}))),
  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
}
