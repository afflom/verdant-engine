definition{
  name: countInvalidTensorKernels,
  parameters: [items : (List Tensor::TensorKernelSpec)],
  result: Nat,
  body: (match items
    | List.nil => 0
    | List.cons item rest => ((if (Tensor::tensorKernelValid item) then 0 else 1) + (countInvalidTensorKernels rest))),
  recursive_argument: items
}

definition{
  name: invalidTensorKernels,
  parameters: [],
  result: Nat,
  body: (countInvalidTensorKernels (Tensor::unifiedTensorKernels))
}

definition{
  name: disorderedStreamingLods,
  parameters: [],
  result: Nat,
  body: (if (Tensor::lodsMonotonic) then 0 else 1)
}

definition{
  name: sameNat,
  parameters: [a : Nat, b : Nat],
  result: Nat,
  body: (prim subtract 1 ((prim subtract a b : Nat) + (prim subtract b a : Nat)) : Nat)
}

definition{
  name: smallerNat,
  parameters: [a : Nat, b : Nat],
  result: Nat,
  body: (prim subtract a (prim subtract a b : Nat) : Nat)
}

definition{
  name: goalsOf,
  parameters: [user : Nat, goals : (List Foundation.Experience.V1.Model::Goal)],
  result: Nat,
  body: (match goals
    | List.nil => 0
    | List.cons goal rest => ((sameNat (project goal user) user) + (goalsOf user rest))),
  recursive_argument: goals
}

definition{
  name: fewestGoals,
  parameters: [
    user : Nat,
    groups : (List Foundation.Experience.V1.Model::UserGroup),
    goals : (List Foundation.Experience.V1.Model::Goal)
  ],
  result: Nat,
  body: (match groups
    | List.nil => 0
    | List.cons group rest => (match rest
      | List.nil => (goalsOf user goals)
      | List.cons next others => (smallerNat (goalsOf user goals) (fewestGoals (user + 1) rest goals)))),
  recursive_argument: groups
}

definition{
  name: journeysPerStakeholder,
  parameters: [],
  result: Nat,
  body: (fewestGoals 0 (Users::groups) (Users::goals))
}

  name: magicConstantViolations,

}

definition{

}

definition{
  name: countUnnumberedSubsystems,
  parameters: [pos : Nat, items : (List Subsystems::SubsystemSpec)],
  result: Nat,
  body: (match items
    | List.nil => 0
    | List.cons item rest => ((if (((project item index) =? pos) && ((project item stratum) <=? (Subsystems::sSeven))) then 0 else 1) + (countUnnumberedSubsystems (pos + 1) rest))),
  recursive_argument: items
}

definition{
  name: unverifiedEngineSubsystems,

  body: ((countUnnumberedSubsystems 0 (Subsystems::subsystems)) + (if (Subsystems::runtimeInvariantsDirect) then 0 else 1))
}
