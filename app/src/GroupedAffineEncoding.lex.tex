definition{
  name: evaluate,
  parameters: [equation : Equation, bits : Nat],
  result: Terms,
  body: (match equation
    | Equation.sumSquares sample sum => (record Terms {first := (ScalarF32::add sum (ScalarF32::multiply sample sample)), second := 0, third := 0})
    | Equation.rms sum => (record Terms {first := (ScalarF32::squareRoot (ScalarF32::divide sum 1107296256)), second := 0, third := 0})
    | Equation.weight sample rms => (let absolute : Nat := (ScalarF32::magnitude sample); (record Terms {first := (if (bits =? 2) then absolute else (ScalarF32::add rms absolute)), second := 0, third := 0}))
    | Equation.moments sample weight total weighted => (record Terms {first := (ScalarF32::add total weight), second := (ScalarF32::add weighted (ScalarF32::multiply weight sample)), third := 0})
    | Equation.initial maximum minimum limit => (let inverse : Nat := (ScalarF32::divide (PackedWord::magnitudeBits limit) (ScalarF32::add maximum (ScalarF32::negate minimum))); (record Terms {first := inverse, second := (ScalarF32::divide 1065353216 inverse), third := 0}))
    | Equation.trial maximum minimum trial limit => (let offset : Nat := (ScalarF32::add (if (bits =? 4) then 3212836864 else 3204448256) (ScalarF32::multiply 1036831949 (PackedWord::magnitudeBits trial))); (record Terms {first := (ScalarF32::divide (ScalarF32::add offset (PackedWord::magnitudeBits limit)) (ScalarF32::add maximum (ScalarF32::negate minimum))), second := 0, third := 0}))
    | Equation.weighted sample weight code total squares products => (let level : Nat := (PackedWord::magnitudeBits code); (let scaled : Nat := (ScalarF32::multiply weight level); (record Terms {first := (ScalarF32::add total scaled), second := (ScalarF32::add squares (ScalarF32::multiply scaled level)), third := (ScalarF32::add products (ScalarF32::multiply scaled sample))})))
    | Equation.determinant weight level square => (record Terms {first := (ScalarF32::add (ScalarF32::multiply weight square) (ScalarF32::negate (ScalarF32::multiply level level))), second := 0, third := 0})
    | Equation.solve weight sample level square product det => (record Terms {first := (ScalarF32::divide (ScalarF32::add (ScalarF32::multiply weight product) (ScalarF32::negate (ScalarF32::multiply sample level))) det), second := (ScalarF32::divide (ScalarF32::add (ScalarF32::multiply square sample) (ScalarF32::negate (ScalarF32::multiply level product))) det), third := 0})
    | Equation.origin product square => (record Terms {first := (ScalarF32::divide product square), second := 0, third := 0})
    | Equation.error sample weight code scale minimum prior => (let delta : Nat := (ScalarF32::add (ScalarF32::add (ScalarF32::multiply scale (PackedWord::magnitudeBits code)) minimum) (ScalarF32::negate sample)); (let loss : Nat := (if (bits =? 2) then (ScalarF32::magnitude delta) else (ScalarF32::multiply delta delta)); (record Terms {first := (ScalarF32::add prior (ScalarF32::multiply weight loss)), second := 0, third := 0})))
    | Equation.stored sample scale offset limit => (record Terms {first := (GroupedAffineScales::roundedLevel (ScalarF32::divide (ScalarF32::add sample offset) scale) limit 1262485504), second := 0, third := 0})),
  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
